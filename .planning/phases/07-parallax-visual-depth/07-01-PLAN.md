'use client';

import { useRef } from 'react';
import { useGSAP } from '@gsap/react';
import { gsap, SplitText } from './gsap-registration';

interface HeroEntranceProps {
  children: React.ReactNode;
}

export function HeroEntrance({ children }: HeroEntranceProps) {
  const containerRef = useRef<HTMLDivElement>(null);

  useGSAP(
    () => {
      const mm = gsap.matchMedia();

      mm.add('(prefers-reduced-motion: no-preference)', () => {
        const container = containerRef.current;
        if (!container) return;

        // Query hero elements
        const orbs = container.querySelectorAll('.h-hero-orb');
        const badge = container.querySelector('.h-badge');
        const h1 = container.querySelector('h1');
        const desc = container.querySelector('.h-hero-desc');
        const actions = container.querySelectorAll('.h-hero-actions > *');

        if (!h1) return;

        // Split headline into words for clip-path reveal
        const split = SplitText.create(h1, { type: 'words' });

        // Build the entrance timeline
        const tl = gsap.timeline({ defaults: { ease: 'power3.out' } });

        // 1. Orbs scale in (0s)
        tl.from(orbs, {
          scale: 0.8,
          opacity: 0,
          duration: 0.8,
          ease: 'power2.out',
          stagger: 0.1,
        }, 0);

        // 2. Badge fades up (0.15s)
        if (badge) {
          tl.from(badge, {
            y: 24,
            opacity: 0,
            duration: 0.5,
          }, 0.15);
        }

        // 3. Headline words reveal via clip-path (0.3s)
        tl.from(split.words, {
          clipPath: 'inset(0 100% 0 0)',
          duration: 0.6,
          stagger: 0.05,
        }, 0.3);

        // 4. Description fades up (0.7s)
        if (desc) {
          tl.from(desc, {
            y: 24,
            opacity: 0,
            duration: 0.5,
          }, 0.7);
        }

        // 5. CTA buttons arrive with spring (0.9s)
        if (actions.length) {
          tl.from(actions, {
            y: 24,
            opacity: 0,
            duration: 0.5,
            ease: 'back.out(1.7)',
            stagger: 0.1,
          }, 0.9);
        }

        // useGSAP context auto-reverts SplitText + kills timeline on unmount
      });
    },
    { scope: containerRef }
  );

  return (
    <div ref={containerRef}>
      {children}
    </div>
  );
}

**Key design decisions:**

1. **Import changed:** Added `ScrollTrigger` to the import from `gsap-registration`. The hero entrance previously didn't need ScrollTrigger (it's a page-load timeline), but parallax requires it for scrub-based scroll linking.

2. **Separate matchMedia block for parallax.** The entrance animation is gated behind `(prefers-reduced-motion: no-preference)` only. The parallax block adds `(min-width: 769px)` because hero orbs are `display: none` at mobile. Two separate `mm.add()` calls within the same `gsap.matchMedia()` instance -- this is the established GSAP pattern for different media conditions within one component.

3. **`gsap.to()` on `y` only.** The entrance uses `gsap.from()` on `scale` and `opacity`. Parallax uses `gsap.to()` on `y`. These animate orthogonal properties so they coexist without conflict. The entrance plays once on load; parallax scrubs continuously with scroll.

4. **`start: 'top top'` for hero.** The hero is above the fold -- it's visible at page load. `top top` means parallax begins when the hero's top edge is at the viewport top (i.e., immediately). `end: 'bottom top'` means it ends when the hero's bottom edge reaches the viewport top (i.e., fully scrolled past). This covers the entire scroll range the hero is visible.

5. **Differential displacement.** Orb 1 moves -50px (slower = farther back), orb 2 moves -80px (faster = closer to viewer). This creates depth separation between the two orbs themselves, reinforcing the layered illusion.

6. **`ease: 'none'`** ensures linear mapping to scroll position. With scrub, non-linear easing would create jerky movement.

7. **`scrub: 0.5`** provides 0.5s of smoothing for natural catch-up, matching the scroll progress bar's scrub pattern from Phase 6.

- The entrance timeline code is completely unchanged from Phase 5 (Plan 05-01). Not a single line modified.
- `useGSAP` context handles cleanup of both the entrance timeline and the parallax ScrollTriggers on unmount.
- `gsap.matchMedia()` auto-reverts parallax animations if the viewport resizes below 769px.

   - Services, Process, Pricing, FAQ sections: accent parallax only
   - Outcomes (dark) section: glow parallax only
   - Final CTA section: final glow parallax only
   - Metrics, AI Callout sections: no parallax (no decorative elements)

3. **Alternating displacement for accents.** `i % 2 === 0 ? -25 : -18` provides subtle variation between diamond and ring accents within the same section. This prevents decorative elements from moving in lockstep, which would look mechanical.

4. **Glow displacement respects overflow:hidden.** The dark section (`.h-dark-inner`) has `overflow: hidden`. Glow 1 at -35px and glow 2 at -25px are conservative enough to stay within the padded container bounds. Similarly, `.h-final` has `overflow: hidden` and -25px is safe.
